  <script>
    // CUSTOMIZE HERE: Change '1511' to your desired 4-digit passcode
    const CORRECT_PASSCODE = "1216";
    let enteredCode = "";

    function updateDisplay() {
      let display = "";
      for (let i = 0; i < 4; i++) {
        display += enteredCode[i] ? "◆ " : "_ ";
      }
      document.getElementById("passcodeDisplay").textContent = display.trim();
    }

    function pressKey(num) {
      if (enteredCode.length < 4) {
        enteredCode += num;
        updateDisplay();
        document.getElementById("error-msg").textContent = "";
      }
    }

    function clearCode() {
      enteredCode = "";
      updateDisplay();
      document.getElementById("error-msg").textContent = "";
    }

    function checkPasscode() {
      if (enteredCode === CORRECT_PASSCODE) {
        document.getElementById("lock-screen").classList.remove("active");
        document.getElementById("gifts-screen").classList.add("active");
        launchRoyalParticles();
      } else {
        document.getElementById("error-msg").textContent = "Access Denied — Check your key code & try again 🔑";
        enteredCode = "";
        updateDisplay();
      }
    }

    function openModal(id) {
      document.getElementById(id).style.display = "flex";
    }

    function closeModal(id) {
      document.getElementById(id).style.display = "none";
    }

    function redeemVoucher(el) {
      el.classList.toggle("claimed");
      const status = el.querySelector(".redeem-status");
      status.textContent = el.classList.contains("claimed") ? "UNLOCKED ✅" : "Activate Pass";
    }

    // Subtle floating royal sparkles & motor emblems
    function launchRoyalParticles() {
      const container = document.getElementById("ambientBg");
      const icons = ["✨", "💎", "👑", "🏎️", "🏍️", "💠"];
      setInterval(() => {
        const el = document.createElement("div");
        el.className = "floating-particle";
        el.innerHTML = icons[Math.floor(Math.random() * icons.length)];
        el.style.left = Math.random() * 100 + "vw";
        el.style.animationDuration = (Math.random() * 4 + 5) + "s";
        container.appendChild(el);
        setTimeout(() => el.remove(), 9000);
      }, 750);
    }
  </script>
